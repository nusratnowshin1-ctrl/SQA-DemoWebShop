pm.test("Status code is 200", function () {
    pm.response.to.have.status(200);
});

const user = pm.response.json();
pm.collectionVariables.set("originalPhone", user.phone);

pm.test("Correct user ID returned", function () {
    pm.expect(user.id).to.eql(
        Number(pm.collectionVariables.get("userId"))
    );
});

pm.test("User name is not empty", function () {
    pm.expect(user.name).to.not.be.empty;
});

pm.test("User email is not empty", function () {
    pm.expect(user.email).to.not.be.empty;
});
