
pm.test("Status code should be 200", function () {
    pm.response.to.have.status(200);
});

const users = pm.response.json();

pm.test("Response should be a non-empty array", function () {
    pm.expect(users).to.be.an("array");
    pm.expect(users.length).to.be.above(0);
});

pm.test("Each user should have ID, Name and Email", function () {
    users.forEach(function (user) {
        pm.expect(user.id).to.exist;
        pm.expect(user.name).to.be.a("string").and.not.empty;
        pm.expect(user.email).to.be.a("string").and.not.empty;
    });
});

// Save first user's ID for the next request
if (Array.isArray(users) && users.length > 0) {
    pm.collectionVariables.set("userId", users[0].id);
}
