// Validate status code
pm.test("Status code is 200", function () {
    pm.response.to.have.status(200);
});

const user = pm.response.json();

// Validate returned ID
pm.test("Returned ID matches saved ID", function () {
    pm.expect(user.id).to.eql(
        Number(pm.collectionVariables.get("userId"))
    );
});

// Validate phone is not empty
pm.test("Phone is not empty", function () {
    pm.expect(user.phone).to.be.a("string").and.not.empty;
});

// Validate updated name
pm.test("Returned name matches updated name", function () {
    pm.expect(user.name).to.eql(
        pm.collectionVariables.get("updatedName")
    );
});
